cmake_minimum_required(VERSION 3.20)
project(ultima7_browser LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(U7_BACKEND_MATOYA OFF CACHE BOOL "" FORCE)
set(U7_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../.build/upstream/ultima7-portable/Black Gate" CACHE PATH "Pinned Ultima7_Decomp portable Black Gate source")
add_subdirectory("${U7_SOURCE}" core)
target_include_directories(u7_game PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
foreach(name IN ITEMS convmgr u7font colbuf sysusage actitem sprite main savegame lsgump gumpmgr getpick inv_ov2 itemdrag controls preload)
  set_source_files_properties("${U7_SOURCE}/game/main/${name}.cpp" DIRECTORY "${U7_SOURCE}" PROPERTIES HEADER_FILE_ONLY TRUE)
  target_sources(u7_game PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/generated/${name}.cpp")
endforeach()
set_source_files_properties("${U7_SOURCE}/game/main/palette/creeper.cpp" DIRECTORY "${U7_SOURCE}" PROPERTIES HEADER_FILE_ONLY TRUE)
target_sources(u7_game PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/generated/creeper.cpp")
set_source_files_properties("${U7_SOURCE}/game/sound/u7sound.cpp" DIRECTORY "${U7_SOURCE}" PROPERTIES HEADER_FILE_ONLY TRUE)
target_sources(u7_game PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/generated/u7sound.cpp")
set_source_files_properties("${U7_SOURCE}/game/main/u7.c" DIRECTORY "${U7_SOURCE}" PROPERTIES HEADER_FILE_ONLY TRUE)
target_sources(u7_game PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/generated/u7.c")
set_source_files_properties("${U7_SOURCE}/game/inter/farstr.cpp" DIRECTORY "${U7_SOURCE}" PROPERTIES HEADER_FILE_ONLY TRUE)
target_sources(u7_game PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/generated/farstr.cpp")
set_source_files_properties("${U7_SOURCE}/game/inter/inter.cpp" DIRECTORY "${U7_SOURCE}" PROPERTIES HEADER_FILE_ONLY TRUE)
target_sources(u7_game PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/generated/inter.cpp")
set_source_files_properties("${U7_SOURCE}/game/zevent/u7event.cpp" DIRECTORY "${U7_SOURCE}" PROPERTIES HEADER_FILE_ONLY TRUE)
target_sources(u7_game PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/generated/u7event.cpp")
set_source_files_properties("${U7_SOURCE}/game/zevent/msclick.cpp" DIRECTORY "${U7_SOURCE}" PROPERTIES HEADER_FILE_ONLY TRUE)
target_sources(u7_game PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/generated/msclick.cpp")
set_source_files_properties("${U7_SOURCE}/game/mainmenu/mainmenu.cpp" DIRECTORY "${U7_SOURCE}" PROPERTIES HEADER_FILE_ONLY TRUE)
target_sources(u7_game PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/generated/mainmenu.cpp")
set_source_files_properties("${U7_SOURCE}/game/mainmenu/credits.cpp" DIRECTORY "${U7_SOURCE}" PROPERTIES HEADER_FILE_ONLY TRUE)
target_sources(u7_game PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/generated/credits.cpp")
set_source_files_properties("${U7_SOURCE}/game/endgame/endgame.cpp" DIRECTORY "${U7_SOURCE}" PROPERTIES HEADER_FILE_ONLY TRUE)
target_sources(u7_game PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/generated/endgame.cpp")
set_source_files_properties("${U7_SOURCE}/game/intro/intro.cpp" DIRECTORY "${U7_SOURCE}" PROPERTIES HEADER_FILE_ONLY TRUE)
target_sources(u7_game PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/generated/intro.cpp")
target_compile_options(u7_game PRIVATE -O2 -fexceptions -Wno-narrowing -Wno-constant-conversion)
add_executable(ultima7 browser.c program.cpp scalers.cpp generated/files.c "${U7_SOURCE}/../third_party/nuked-opl3/opl3.c")
target_include_directories(ultima7 PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${U7_SOURCE}/platform" "${U7_SOURCE}/../third_party/nuked-opl3")
target_link_libraries(ultima7 PRIVATE u7_game)
target_compile_options(ultima7 PRIVATE -O2 -fexceptions)
target_link_options(ultima7 PRIVATE -O1 -fexceptions
  "-sASYNCIFY=1" "-sASYNCIFY_STACK_SIZE=2097152" "-sSTACK_SIZE=2097152"
  "-sINITIAL_MEMORY=67108864" "-sALLOW_MEMORY_GROWTH=1"
  "-sMODULARIZE=1" "-sEXPORT_NAME=createUltima7" "-sENVIRONMENT=web,node"
  "-sFORCE_FILESYSTEM=1" "-sEXIT_RUNTIME=0" "-sASSERTIONS=1" "-sEMULATE_FUNCTION_POINTER_CASTS=1"
  "-sEXPORTED_RUNTIME_METHODS=['FS','IDBFS','HEAPU8','HEAPF32','callMain','ccall']"
  "-sEXPORTED_FUNCTIONS=['_main','_malloc','_free','_web_key','_web_mouse','_web_select_answer','_web_continue','_web_menu_command','_web_set_audio_modes','_web_frame_ptr','_web_palette_ptr','_web_frame_serial','_web_audio','_web_save_sync','_web_audio_finished','_web_current_music','_web_restart_music','_web_scale_frame','_web_set_audio_state','_web_audio_state','_web_avatar_position']"
  "-lidbfs.js")
set_target_properties(ultima7 PROPERTIES SUFFIX ".js" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../dist/engine")
option(U7_INTERACTION_PROBE "Build isolated interaction regression engine" OFF)
if(U7_INTERACTION_PROBE)
  set_source_files_properties("${CMAKE_CURRENT_SOURCE_DIR}/generated/main.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/generated/getpick.cpp" DIRECTORY "${U7_SOURCE}" PROPERTIES COMPILE_DEFINITIONS U7_INTERACTION_PROBE)
  target_sources(u7_game PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/interaction-probe.cpp")
  set_target_properties(ultima7 PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../.local/test-engine")
endif()
